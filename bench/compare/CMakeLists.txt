# Comparison programs (zen_platform against GLFW 3 and SDL2): built only with -DPLATFORM_BUILD_COMPARE=ON, so the
# normal build needs neither. See tools/bench_compare.sh and docs/COMPARACAO_CORE.md.
#   -DZP_COMPARE_GLFW_ROOT=<dir>  a prefix holding include/GLFW and lib/ (default: what pkg-config / the system has)
# With PLATFORM_LINUX_BACKEND=wayland the programs are built without X11 (BENCH_WAYLAND): the modes that need no input
# injection (start-up, memory, idle, cycle, wait), against the small compositor of the tests run as a program of its own
# (wl_compositor), through tools/bench_wayland.sh. The event modes (bench_run, XTest) are X11 only.
find_package(PkgConfig REQUIRED)
find_package(Threads REQUIRED)
if(PLATFORM_BACKEND_WAYLAND)
    set(BENCH_DEFS BENCH_WAYLAND)
    set(BENCH_X11 "")
else()
    find_package(X11 REQUIRED)
    set(BENCH_DEFS "")
    set(BENCH_X11 X11)
endif()

add_executable(bench_zp bench_main.c bench_zp.c)
target_include_directories(bench_zp PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_definitions(bench_zp PRIVATE ${BENCH_DEFS})
target_link_libraries(bench_zp PRIVATE platform ${BENCH_X11} Threads::Threads)

if(PLATFORM_BACKEND_WAYLAND)
    # sets BENCH_T0 (the monotonic clock) and runs the program: what bench_run does for the X11 programs
    add_executable(bench_once bench_once.c)
    pkg_check_modules(WAYLAND_SERVER IMPORTED_TARGET wayland-server)
    if(WAYLAND_SERVER_FOUND AND NOT BUILD_SHARED_LIBS)
        add_executable(wl_compositor wl_compositor.c)
        add_dependencies(wl_compositor wayland_server_headers)
        target_include_directories(wl_compositor PRIVATE ${WAYLAND_GEN_DIR} ${CMAKE_SOURCE_DIR}/include)
        if(WAYLAND_POINTER_LOCK)
            target_compile_definitions(wl_compositor PRIVATE ZEN_WL_POINTER_LOCK)
        endif()
        target_link_libraries(wl_compositor PRIVATE platform PkgConfig::WAYLAND_SERVER)
    else()
        message(STATUS "compare: wayland-server not found (or a shared build), wl_compositor is not built")
    endif()
else()
    add_executable(bench_run bench_run.c)
    target_link_libraries(bench_run PRIVATE X11 Xtst)
endif()

add_library(mcount SHARED mcount.c)
set_target_properties(mcount PROPERTIES PREFIX "lib")
target_link_libraries(mcount PRIVATE ${CMAKE_DL_LIBS})

# GLFW: the distribution's shared library, found by path so that a copy unpacked in a directory of its own works too
if(ZP_COMPARE_GLFW_ROOT)
    find_library(ZP_GLFW_LIB glfw PATHS ${ZP_COMPARE_GLFW_ROOT}/lib ${ZP_COMPARE_GLFW_ROOT}/usr/lib/x86_64-linux-gnu NO_DEFAULT_PATH)
    find_path(ZP_GLFW_INC GLFW/glfw3.h PATHS ${ZP_COMPARE_GLFW_ROOT}/include ${ZP_COMPARE_GLFW_ROOT}/usr/include NO_DEFAULT_PATH)
else()
    find_library(ZP_GLFW_LIB glfw)
    find_path(ZP_GLFW_INC GLFW/glfw3.h)
endif()
if(ZP_GLFW_LIB AND ZP_GLFW_INC)
    add_executable(bench_glfw bench_main.c bench_glfw.c)
    target_include_directories(bench_glfw PRIVATE ${CMAKE_CURRENT_SOURCE_DIR} ${ZP_GLFW_INC})
    target_compile_definitions(bench_glfw PRIVATE ${BENCH_DEFS})
    target_link_libraries(bench_glfw PRIVATE ${ZP_GLFW_LIB} ${BENCH_X11} Threads::Threads)
else()
    message(STATUS "compare: GLFW not found (set ZP_COMPARE_GLFW_ROOT), bench_glfw is not built")
endif()

pkg_check_modules(ZP_SDL2 sdl2)
if(ZP_SDL2_FOUND)
    add_executable(bench_sdl2 bench_main.c bench_sdl2.c)
    target_include_directories(bench_sdl2 PRIVATE ${CMAKE_CURRENT_SOURCE_DIR} ${ZP_SDL2_INCLUDE_DIRS})
    target_compile_definitions(bench_sdl2 PRIVATE ${BENCH_DEFS})
    target_link_libraries(bench_sdl2 PRIVATE ${ZP_SDL2_LINK_LIBRARIES} ${BENCH_X11} Threads::Threads)
else()
    message(STATUS "compare: SDL2 not found, bench_sdl2 is not built")
endif()
